clients_db=# -- A. the full chain since 17:00 (triggers → page items → escalations → any writer rebuilds)
SELECT s.domain, w.item_type, coalesce(w.spec->>'page_name', w.spec->>'function') AS what,
       w.status, (w.error IS NOT NULL AND w.error <> '') AS has_err, w.updated_at
FROM site_work_items w JOIN sites s ON s.id = w.site_id
WHERE w.updated_at > '2026-07-03 17:00'
  AND ( w.item_key LIKE 'f8_rerender_briefexp:%'
     OR (w.item_type = 'page_rerender' AND w.spec->>'reason' = 'section_data_resolved')
     OR w.item_type = 'needs_page' )
ORDER BY w.updated_at;

-- B. the word in context (echo-contamination vs ordinary English)
SELECT s.domain, p.name,
       substring(lower(pc.rendered_html) FROM '.{0,70}gauntlet.{0,70}') AS ctx
FROM page_components pc JOIN content_components cc ON cc.id = pc.component_id
JOIN pages p ON p.id = pc.page_id JOIN sites s ON s.id = p.site_id
WHERE cc.function = 'brief-explanation' AND strpos(lower(pc.rendered_html),'gauntlet') > 0
ORDER BY 1, 2;

-- C. which content_data FIELD carries it (the decider for llm-copy contamination)
SELECT s.domain, p.name, t.key, left(t.value, 90) AS val
FROM page_components pc JOIN content_components cc ON cc.id = pc.component_id
JOIN pages p ON p.id = pc.page_id JOIN sites s ON s.id = p.site_id,
LATERAL jsonb_each_text(pc.content_data) t
WHERE cc.function = 'brief-explanation' AND lower(t.value) LIKE '%gauntlet%'
ORDER BY 1, 2, 3;
