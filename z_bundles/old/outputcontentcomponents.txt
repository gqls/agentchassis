SELECT cc.id::text AS component_id, cc.function, cc.component_level, cc.is_active,
       cc.forked_from IS NOT NULL AS is_fork, length(cc.html_template) AS html_len
FROM content_components cc
JOIN page_components pc ON pc.component_id = cc.id
JOIN pages p ON pc.page_id = p.id
WHERE p.site_id = 'e33263f4-74f8-494f-b191-546845dbbddf'::uuid
  AND p.name = 'game-economy-simulator'
ORDER BY cc.function;
