clients_db=# -- W8 step 2 (read-only): the image question, without guessing.

-- 2.1 Edit B's query BY HAND for page 'index' — the decisive data-side check. Rows here
--     mean the data matches and any persistent has_image=f points at the code path/pod;
--     0 rows means the filter mismatch is in the data and is visible right here.
SELECT spi.kind, spi.key, left(a.url, 70) AS url_head
FROM site_plan_imagery spi
JOIN site_plans sp ON sp.id = spi.plan_id AND sp.is_current = true
JOIN assets a ON a.site_id = sp.site_id
             AND a.asset_key = spi.key
             AND a.status = 'active'
WHERE sp.site_id = (SELECT id FROM sites WHERE domain = 'idea.uk')
  AND spi.scope = 'section'
  AND spi.scope_ref LIKE 'index' || ':%'
  AND spi.kind IN ('illustration', 'icon', 'infographic')
ORDER BY spi.kind, spi.ordering;

-- 2.2 Provenance: which rebuilds produced the current renders (identifies whether the
--     section_back=t render ran pre- or post-deploy, and watches the two w8 items):
SELECT item_key, created_by, status, created_at, updated_at
FROM site_work_items
WHERE site_id = (SELECT id FROM sites WHERE domain='idea.uk')
  AND item_type = 'needs_page'
  AND created_at > now() - interval '30 hours'
ORDER BY created_at;

-- 2.3 Re-run after the two w8 items complete:
SELECT p.name,
       (pc.rendered_html LIKE '%data-component="brief-explanation"%') AS section_back,
       (pc.rendered_html LIKE '%illustration_%')                       AS has_image,
       pc.updated_at
FROM page_components pc JOIN pages p ON p.id = pc.page_id
WHERE p.site_id = (SELECT id FROM sites WHERE domain='idea.uk')
  AND p.name IN ('index','tools') AND pc.slot_name = 'brief-explanation'
ORDER BY p.name;
